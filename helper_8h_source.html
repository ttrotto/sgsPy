172template <typename T>
173inline T
174getPixelValueDependingOnType(
175 GDALDataType type,
176 void *p_data,
177 size_t index
178) {
179 switch (type) {
180 case GDT_Int8:
181 return static_cast<T>(((int8_t *)p_data)[index]);
182 case GDT_UInt16:
183 return static_cast<T>(((uint16_t *)p_data)[index]);
184 case GDT_Int16:
185 return static_cast<T>(((int16_t *)p_data)[index]);
186 case GDT_UInt32:
187 return static_cast<T>(((uint32_t *)p_data)[index]);
188 case GDT_Int32:
189 return static_cast<T>(((int32_t *)p_data)[index]);
190 case GDT_Float32:
191 return static_cast<T>(((float *)p_data)[index]);
192 case GDT_Float64:
193 return static_cast<T>(((double *)p_data)[index]);
194 default:
195 throw std::runtime_error("raster pixel data type not supported.");
196 }

375inline GDALDataset *
376createDataset(
377 std::string filename,
378 std::string driverName,
379 int width,
380 int height,
381 double *geotransform,
382 std::string projection,
383 RasterBandMetaData *bands,
384 size_t bandCount,
385 bool useTiles,
386 std::map<std::string, std::string>& driverOptions)
387{
388 GDALDriver *p_driver = GetGDALDriverManager()->GetDriverByName(driverName.c_str());
389 if (!p_driver) {
390 throw std::runtime_error("unable to find dataset driver.");
391 }
392
393 char **papszOptions = nullptr;
394 if (useTiles) {
395 if (driverOptions.find("TILED") == driverOptions.end()) {
396 papszOptions = CSLSetNameValue(papszOptions, "TILED", "YES");
397 }
398 if (driverOptions.find("BLOCKXSIZE") == driverOptions.end()) {
399 const char *xBlockSizeOption = std::to_string(bands[0].xBlockSize).c_str();
400 papszOptions = CSLSetNameValue(papszOptions, "BLOCKXSIZE", xBlockSizeOption);
401 }
402 if (driverOptions.find("BLOCKYSIZE") == driverOptions.end()) {
403 const char *yBlockSizeOption = std::to_string(bands[0].yBlockSize).c_str();
404 papszOptions = CSLSetNameValue(papszOptions, "BLOCKYSIZE", yBlockSizeOption);
405 }
406 }
407
408 for (auto const& [key, val] : driverOptions) {
409 papszOptions = CSLSetNameValue(papszOptions, key.c_str(), val.c_str());
410 }
411
412 GDALDataset *p_dataset = p_driver->Create(
413 filename.c_str(),
414 width,
415 height,
416 bandCount,
417 bands[0].type,
418 papszOptions
419 );
420 CSLDestroy(papszOptions);
421
422 if (!p_dataset) {
423 throw std::runtime_error("unable to create dataset with driver.");
424 }
425
426 CPLErr err = p_dataset->SetGeoTransform(geotransform);
427 if (err) {
428 throw std::runtime_error("error setting geotransform.");
429 }
430
431 err = p_dataset->SetProjection(projection.c_str());
432 if (err) {
433 throw std::runtime_error("error setting projection.");
434 }
435
436 for (size_t band = 0; band < bandCount; band++) {
437 GDALRasterBand *p_band = p_dataset->GetRasterBand(band + 1);
438 p_band->SetDescription(bands[band].name.c_str());
439 p_band->SetNoDataValue(bands[band].nan);
440 p_band->GetBlockSize(&bands[band].xBlockSize, &bands[band].yBlockSize);
441 bands[band].p_band = p_band;
442 }
443
444 return p_dataset;
445}

648inline void
649rasterBandIO(
650 RasterBandMetaData& band,
651 void *p_buffer,
652 int xBlockSize,
653 int yBlockSize,
654 int xBlock,
655 int yBlock,
656 int xValid,
657 int yValid,
658 bool read,
659 bool threaded = true)
660{
661 CPLErr err;
662 bool useBlock = xBlockSize == band.xBlockSize &&
663 yBlockSize == band.yBlockSize;
664
665 if (threaded) {
666 band.p_mutex->lock();
667 }
668 if (useBlock && read) {
669 err = read ?
670 band.p_band->ReadBlock(xBlock, yBlock, p_buffer) :
671 band.p_band->WriteBlock(xBlock, yBlock, p_buffer);
672 }
673 else {
674 err = band.p_band->RasterIO(
675 read ? GF_Read : GF_Write,
676 xBlock * xBlockSize,
677 yBlock * yBlockSize,
678 xValid,
679 yValid,
680 p_buffer,
681 xValid, //buffer size must equal the window size, otherwise GDAL resamples
682 yValid,
683 band.type,
684 band.size,
685 static_cast<GSpacing>(band.size) * xBlockSize //rows in the buffer stay xBlockSize wide, like ReadBlock()
686 );
687 }
688 if (threaded) {
689 band.p_mutex->unlock();
690 }
691 if (err) {
692 throw read ?
693 std::runtime_error("unable to read block from raster.") :
694 std::runtime_error("unable to write block to raster.");
695 }

838class Variance {
839 private:
840 int64_t k = 0; //count
841 double M = 0; //running mean
842 double S = 0; //sum of squares
843 double oldM = 0;
844
845 public:
846 inline void
852 update(double x) {
853 k++;
854 oldM = M;
855
856 //update running mean
857 M = M + (x - M) / static_cast<double>(k);
858
859 //update sum of squares
860 S = S + (x - M) * (x - oldM);
861 }
862
868 inline double
869 getMean() {
870 return M;
871 }
872
879 inline double
880 getStdev() {
881 double variance = S / static_cast<double>(k);
882 return std::sqrt(variance);
883 }
884
890 inline int64_t
891 getCount() {
892 return this->k;
893 }
894};

945inline uint64_t
946getProbabilityMultiplier(double width, double height, double pixelWidth, double pixelHeight, int startMult, int numSamples, bool useMindist, double accessibleArea) {
947 double numer = static_cast<double>(numSamples * startMult * (useMindist ? 3 : 1));
948 double denom = height * width;
949
950 if (accessibleArea != -1) {
951 double totalArea = width * pixelWidth * height * pixelHeight;
952 numer *= (totalArea / accessibleArea);
953 }
954
955 if (numer > denom) {
956 return 0;
957 }
958
959 uint8_t bits = static_cast<uint8_t>(std::ceil(std::log2(denom) - std::log2(numer)));
960 return (1 << bits) - 1;
961}

982class RandValController {
983private:
984 std::vector<bool> randVals;
985 size_t randValIndex = 0;
986 uint64_t multiplier = 0;
987 xso::xoshiro_4x64_plus *p_rng = nullptr;
988 bool alwaysTrue = false;
989
990public:
1000 RandValController(int xBlockSize, int yBlockSize, uint64_t multiplier, xso::xoshiro_4x64_plus *p_rng) {
1001 if (multiplier == 0) {
1002 this->alwaysTrue = true;
1003 }
1004 else {
1005 this->randVals.resize(xBlockSize * yBlockSize);
1006 this->randValIndex = static_cast<size_t>(xBlockSize * yBlockSize);
1007 this->multiplier = multiplier;
1008 this->p_rng = p_rng;
1009 }

1024 inline void
1025 calculateRandValues(void) {
1026 if (alwaysTrue) {
1027 return;
1028 }
1029
1030 for (size_t i = 0; i < randValIndex; i++) {
1031 randVals[i] = (((*p_rng)() >> 11) & multiplier) == multiplier;
1032 }
1033 randValIndex = 0;
1034 }
1035
1039 inline bool
1040 next(void) {
1041 if (alwaysTrue) {
1042 return true;
1043 }
1044
1045 bool retval = randVals[randValIndex];
1046 randValIndex++;
1047 return retval;
1048 }

1078inline bool is_valid_sample(double x, double y, NeighborMap& neighbor_map, float mindist, float mindist_sq) {
1079 int cx = static_cast<int>(std::floor(x / mindist));
1080 int cy = static_cast<int>(std::floor(y / mindist));
1081
1082 for (int dx = -1; dx <= 1; dx++) {
1083 for (int dy = -1; dy <= 1; dy++) {
1084 std::pair<int, int> neighbor = {cx + dx, cy + dy};
1085
1086 auto it = neighbor_map.find(neighbor);
1087 if (it == neighbor_map.end())
1088 continue;
1089
1090 for (const auto &[nx, ny] : it->second) {
1091 float dxp = x - nx;
1092 float dyp = y - ny;
1093 float dist_sq = dxp * dxp + dyp * dyp;
1094
1095 if (dist_sq < mindist_sq) {
1096 return false;
1097 }
1098 }
1099 }
1100 }
1101
1102 // add point to neighborhood
1103 neighbor_map[{cx, cy}].emplace_back(x, y);
1104 return true;
1105}
